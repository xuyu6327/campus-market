```json
// POST /api/v1/admin/auth/login

// 审核管理
GET    /api/v1/admin/audit/queue
POST   /api/v1/admin/audit/{id}/approve
POST   /api/v1/admin/audit/{id}/reject    // { "reason": "含广告" }

// 举报处理
GET    /api/v1/admin/reports?status=0&type=&page=&size=
POST   /api/v1/admin/reports/{id}/handle
// Request: { "action":"approve", "penalty":2, "result":"下架处理" }
// penalty: 1警告 2下架扣分 3限发 4封号

// 申诉处理
GET    /api/v1/admin/appeals
POST   /api/v1/admin/appeals/{id}/handle  // { "action":"approve|reject" }

// 敏感词管理 CRUD + 批量导入
// 数据统计 GET /api/v1/admin/statistics/overview?period=week
```

> **统计实现方案（V1.0）**：
> - **实时统计**（用户数、商品数、在售数等计数型指标）：直接 `COUNT` 查询，结果缓存 Redis 5 分钟
> - **聚合统计**（日/周/维度的注册量、发布量、成交量等）：每天凌晨 1 点跑定时任务，将前一天数据聚合写入 `daily_statistics` 表，查询时直接读聚合表
> - **热门搜索词**：Redis ZSET `search:hotwords` 实时累加，每小时刷新 Top 10
> - **信用分分布**：`SELECT FLOOR(credit_score/10)*10 AS range, COUNT(*) FROM sys_user GROUP BY range`
>
> ```sql
> -- 每日聚合统计表（V1.0 可选，不强制）
> CREATE TABLE daily_statistics (
>     stat_date        DATE         PRIMARY KEY,
>     new_users        INT          DEFAULT 0,
>     new_goods        INT          DEFAULT 0,
>     booked_count     INT          DEFAULT 0,
>     confirmed_count  INT          DEFAULT 0,
>     released_count   INT          DEFAULT 0,
>     report_count     INT          DEFAULT 0,
>     created_at       DATETIME     DEFAULT CURRENT_TIMESTAMP
> ) COMMENT '每日统计聚合表';
> ```
// 操作日志 GET /api/v1/admin/logs (超管)
```

### 2.2 新增索引 SQL

```sql
-- 商品列表页最常用复合查询： WHERE category_id=? AND status=1 AND price BETWEEN ? AND ? ORDER BY created_at DESC
-- created_at放在最后避免filesort
ALTER TABLE goods_info ADD INDEX idx_cat_status_price_time (category_id, status, price, created_at);

-- 我的发布： WHERE user_id=? AND status=? ORDER BY created_at DESC

@PostMapping("/auth/login")
public Result login(@RequestBody LoginRequest req) {
    // 1. 查用户 by HMAC盲索引（不使用AES加密值做WHERE查询）
    // ★ 安全方案：phone字段用AES-GCM随机IV加密存储(仅解密展示)，phone_hash字段用HMAC-SHA256做盲索引(用于WHERE等值查询)
    // HMAC密钥与AES密钥分离，查询时 WHERE phone_hash = HMAC(输入手机号)
    String phoneHash = HmacUtil.sha256(req.getPhone());
    User user = userMapper.selectOne(new LambdaQueryWrapper<User>()
        .eq(User::getPhoneHash, phoneHash));
    if (user == null) throw new BizException(401, "手机号或密码错误");
    
    // 检查账号状态
    if (user.getStatus() != 1) {
        throw new BizException(403, user.getStatus() == 0 ? "账号已被禁用" : "账号已注销");
    }
    
    // 2. 锁定检查
    String failKey = "login:fail:" + req.getPhone();
    Integer failCount = (Integer) redis.opsForValue().get(failKey);
    if (failCount != null && failCount >= 5) {
        throw new BizException(429, "错误次数过多，请15分钟后再试");
    }
    
    // 3. 验密码
    if (!BCrypt.checkpw(req.getPassword(), user.getPassword())) {
        // ★ incr后expire：首次incr创建key，每次失败刷新15分钟锁定
        long count = redis.opsForValue().increment(failKey);
        redis.expire(failKey, 15, MINUTES);
        throw new BizException(401, "手机号或密码错误，剩余" + (5-count) + "次");
    }
    
    // 4. 成功 → 清失败 → 签发token
    redis.delete(failKey);
    String token = jwtUtil.generate(user.getUserId());
    redis.set("session:" + token, JSON.toJSONString(user), 30, DAYS);
    return Result.ok(LoginVO.builder().token(token).userId(user.getUserId())
        .nickname(user.getNickname()).avatar(user.getAvatar()).build());

    User user = userMapper.selectById(uid);

    String lockKey = "booking:lock:" + goodsId;
    String lockVal = UUID.randomUUID().toString();
    if (!redis.setIfAbsent(lockKey, lockVal, 30, SECONDS))
        throw new BizException(409, "操作频繁，请稍后再试");

        // DB乐观锁兜底
        GoodsInfo goods = goodsMapper.selectById(goodsId);

        if (goods.getUserId().equals(uid)) throw new BizException(400, "不能预订自己的");

            .eq(GoodsInfo::getGoodsId, goodsId)
            .eq(GoodsInfo::getStatus, 1)
            .eq(GoodsInfo::getVersion, goods.getVersion() - 1));
        if (rows == 0) throw new BizException(409, "商品状态已变更");

        TradeOrder order = new TradeOrder();

        order.setBuyerId(uid);
        order.setSellerId(goods.getUserId());

void change(Long userId, int delta, String reason) {
    // 1. 查询当前分数
    User user = userMapper.selectById(userId);
    int beforeScore = user.getCreditScore();
    // 2. 计算新分数（上限100，下限0）
    int afterScore = Math.max(0, Math.min(100, beforeScore + delta));
    // 3. UPDATE sys_user SET credit_score = afterScore WHERE user_id=?
    user.setCreditScore(afterScore);

        user.setStatus(0);
        userMapper.updateById(user);

    CreditLog log = new CreditLog();
    log.setUserId(userId);
    log.setDelta(delta);
    log.setBeforeScore(beforeScore);
    log.setAfterScore(afterScore);
    log.setReason(reason);
    log.setOperator(0);  // 0=系统
    creditLogMapper.insert(log);
}

@Component
public class BookingTimeoutJob {
    @Scheduled(cron = "0 * * * * ?")  // 每分钟
    public void release() {
        String jobLock = "booking:release-job:lock";
        String lockVal = UUID.randomUUID().toString();
        if (!redis.setIfAbsent(jobLock, lockVal, 120, SECONDS)) return; // 多实例互斥
        
        try {
            Set<String> ids = redis.zrangeByScore("booking:timeout:queue", 0,
                System.currentTimeMillis());
            if (ids == null || ids.isEmpty()) return;
            
            for (String id : ids) {
                try { releaseOne(Long.valueOf(id)); }
                catch (Exception e) { log.error("释放失败 orderId={}", id, e); }
                redis.zrem("booking:timeout:queue", id);
            }
        } finally { redis.execute(unlockScript, List.of(jobLock), lockVal); }
    }
    
    @Transactional
    void releaseOne(Long orderId) {
        TradeOrder order = orderMapper.selectById(orderId);
        if (order == null || order.getStatus() != 1) return; // 幂等
        
        order.setStatus(4); order.setReleasedAt(LocalDateTime.now()); order.setCancelBy(3);
        orderMapper.updateById(order);
        
        GoodsInfo goods = goodsMapper.selectById(order.getGoodsId());
        goods.setStatus(1); goods.setVersion(goods.getVersion() + 1);
        int rows = goodsMapper.update(goods, new LambdaUpdateWrapper<GoodsInfo>()
            .eq(GoodsInfo::getGoodsId, goods.getGoodsId())
            .eq(GoodsInfo::getStatus, 2)              // 必须当前是"已预订"
            .eq(GoodsInfo::getVersion, goods.getVersion() - 1));
        
        if (rows == 0) {
            log.error("释放商品失败(乐观锁冲突): goodsId={}, orderId={}",
                goods.getGoodsId(), orderId);
            throw new BizException(409, "商品状态已变更，释放失败");
        }
        
        creditService.change(order.getBuyerId(), -3, "预订超时释放");
        // ★ creditService.change()内部：若信用分跌至<10则自动冻结账号(user.status=0)
        eventPublisher.publishEvent(new BookingReleasedEvent(order));
    }

### 3.7 7天默认好评 + 30天自动下架 定时任务

```
@Component
public class AutoCompleteJob {
    
    // ===== 7天未评 → 自动默认好评（每天凌晨2点执行） =====
    @Scheduled(cron = "0 0 2 * * ?")
    @Transactional
    public void autoPraise() {
        // 查询：交易完成超过7天但双方都未评价的订单
        LocalDateTime sevenDaysAgo = LocalDateTime.now().minusDays(7);
        
        // 找出所有7天前完成的订单
        List<TradeOrder> orders = orderMapper.selectList(
            new LambdaQueryWrapper<TradeOrder>()
                .eq(TradeOrder::getStatus, 2)
                .lt(TradeOrder::getConfirmedAt, sevenDaysAgo));
        // ★ 不设NOT EXISTS过滤：insertAutoEval内部幂等检查自动跳过已评价的
        
        for (TradeOrder order : orders) {
            // 为买卖双方各插入一条默认好评
            insertAutoEval(order.getOrderId(), order.getBuyerId(), order.getSellerId());
            insertAutoEval(order.getOrderId(), order.getSellerId(), order.getBuyerId());
        }
    }
    
    private void insertAutoEval(Long orderId, Long evaluatorId, Long targetId) {
        // 幂等检查：该用户对该订单是否已评价（支持一方已评另一方未评的场景）
        Long exist = evalMapper.selectCount(new LambdaQueryWrapper<UserEvaluation>()
            .eq(UserEvaluation::getOrderId, orderId)
            .eq(UserEvaluation::getEvaluatorId, evaluatorId));
        if (exist > 0) return;
        
        UserEvaluation eval = new UserEvaluation();
        eval.setOrderId(orderId);
        eval.setEvaluatorId(evaluatorId);
        eval.setTargetId(targetId);
        eval.setRating(1);           // 默认好评
        eval.setIsAuto(1);           // 标记为自动评价
        evalMapper.insert(eval);
        
        creditService.change(targetId, 2, "自动默认好评");
    }
    
    // ===== 30天未售出 → 自动下架（每天凌晨3点执行） =====
    // ★ 使用 last_relisted_at（最近上架时间）计算，而非 created_at
    //   避免曾被预订后释放的商品因 created_at 过旧被立即下架
    @Scheduled(cron = "0 0 3 * * ?")
    public void autoOffline() {
        LocalDateTime thirtyDaysAgo = LocalDateTime.now().minusDays(30);
        goodsMapper.update(null, new LambdaUpdateWrapper<GoodsInfo>()
            .set(GoodsInfo::getStatus, 0)                   // 已下架
            .eq(GoodsInfo::getStatus, 1)                    // 当前在售
            .lt(GoodsInfo::getLastRelistedAt, thirtyDaysAgo)); // 30天前上架（按最近上架时间）
    }
}

### 3.10 DFA 敏感词过滤实现

```
// 使用 DFA（确定有限自动机）算法构建敏感词树，替代纯 LIKE 匹配
// 启动时从 DB 加载所有启用的敏感词，构建 DFA 树存入 JVM 内存
// 每小时从 DB 刷新一次（检测新增/删除的敏感词）

@Component
public class SensitiveWordFilter {
    
    private Map<Character, Object> dfaTree;
    private static final String IS_END = "isEnd";
    
    @PostConstruct
    public void init() {
        reload();
    }
    
    // 每小时刷新
    @Scheduled(fixedRate = 3600000)
    public void reload() {
        List<SensitiveWord> words = sensitiveWordMapper.selectList(
            new LambdaQueryWrapper<SensitiveWord>().eq(SensitiveWord::getStatus, 1));
        dfaTree = buildDfaTree(words);
    }
    
    private Map<Character, Object> buildDfaTree(List<SensitiveWord> words) {
        Map<Character, Object> root = new HashMap<>();
        for (SensitiveWord sw : words) {
            Map<Character, Object> current = root;
            for (char c : sw.getWord().toCharArray()) {
                @SuppressWarnings("unchecked")
                Map<Character, Object> next = (Map<Character, Object>) current.get(c);
                if (next == null) {
                    next = new HashMap<>();
                    current.put(c, next);
                }
                current = next;
            }
            current.put(IS_END, "1");  // 标记词尾
        }
        return root;
    }
    
    // 检查文本是否包含敏感词，返回命中的敏感词列表
    public Set<String> check(String text) {
        Set<String> hits = new HashSet<>();
        for (int i = 0; i < text.length(); i++) {
            int j = i;
            Map<Character, Object> current = dfaTree;
            StringBuilder sb = new StringBuilder();
            while (j < text.length()) {
                char c = text.charAt(j);
                @SuppressWarnings("unchecked")
                Map<Character, Object> next = (Map<Character, Object>) current.get(c);
                if (next == null) break;
                sb.append(c);
                current = next;
                if ("1".equals(current.get(IS_END))) {
                    hits.add(sb.toString());
                }
                j++;
            }
        }
        return hits;
    }
}
